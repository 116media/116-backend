```csharp
// ArticleEntity.cs
public class ArticleEntity : Aggregate<Guid>
{
    public Guid?   CustomerId       { get; private set; }
    public Guid?   OrderItemId      { get; private set; }   // NULL for free content
    public Guid    CategoryId       { get; private set; }
    public string  Title            { get; private set; } = null!;
    public string  Slug             { get; private set; } = null!;
    // Short teaser / aperçu (100–300 chars). Empty string on draft creation, filled in step 2.
    public string  Headline         { get; private set; } = string.Empty;
    // Rich-text body HTML. Empty string on draft creation (step 1), filled in step 2.
    public string  Body             { get; private set; } = string.Empty;
    public string? CoverImageUrl    { get; private set; }
    // Identity user UUID from JWT claims. No FK to identity schema by design.
    // Set by handler from HttpContext.User — never passed by the client.
    public string  AuthorId         { get; private set; } = null!;
    // Stamped by commerce payment verification — never set through article endpoints
    public bool    SocialBoost      { get; private set; }
    public bool    IsPromoted       { get; private set; }
    public DateTimeOffset? PromotedUntil { get; private set; }
    public EnumContentStatus Status  { get; private set; }
    public string? RejectionReason  { get; private set; }
    public DateTimeOffset? PublishedAt { get; private set; }
    public string? MetaTitle         { get; private set; }
    public string? MetaDescription   { get; private set; }

    // Denormalized counters
    public int LikeCount     { get; private set; }
    public int CommentCount  { get; private set; }
    public int ShareCount    { get; private set; }
    public int BookmarkCount { get; private set; }

    // Navigation
    public CustomerEntity?        Customer  { get; private set; }
    public ContentOrderItemEntity? OrderItem { get; private set; }
    public CategoryEntity         Category  { get; private set; } = null!;
    public ICollection<ArticleImageEntity>    Images    { get; } = new List<ArticleImageEntity>();
    public ICollection<ArticleTagEntity>      Tags      { get; } = new List<ArticleTagEntity>();
    public ICollection<ArticleLikeEntity>     Likes     { get; } = new List<ArticleLikeEntity>();
    public ICollection<ArticleBookmarkEntity> Bookmarks { get; } = new List<ArticleBookmarkEntity>();
    public ICollection<ArticleShareEntity>    Shares    { get; } = new List<ArticleShareEntity>();
    public ICollection<ArticleCommentEntity>  Comments  { get; } = new List<ArticleCommentEntity>();

    private ArticleEntity() { }

    // Step 1 — "Save Draft" — free content (no customer, no order)
    // body and headline are empty at creation; filled in step 2 via UpdateContent()
    public static ArticleEntity CreateFree(Guid id, Guid categoryId,
        string title, string slug, string authorId) { ... }

    // Step 1 — "Save Draft" — paid content (always has a customer + order item)
    public static ArticleEntity CreatePaid(Guid id, Guid customerId, Guid orderItemId,
        Guid categoryId, string title, string slug, string authorId) { ... }

    // Step 2 — called by UpdateArticleCommandHandler when admin clicks "Submit"
    // Saves headline, body (with Cloudinary URLs), and optional cover image URL
    public void UpdateContent(string headline, string body, string? coverImageUrl) { ... }

    // Status transitions
    public void Submit()                  => Status = EnumContentStatus.PendingPayment;
    public void MarkPendingReview()       => Status = EnumContentStatus.PendingReview;
    public void Approve()                 => Status = EnumContentStatus.Approved;
    public void Publish()                 { Status = EnumContentStatus.Published; PublishedAt = DateTimeOffset.UtcNow; }
    public void Reject(string reason)     { Status = EnumContentStatus.Rejected; RejectionReason = reason; }
    public void Archive()                 => Status = EnumContentStatus.Archived;

    // Commerce stamps — called by payment verification flow, never by article endpoints
    public void StampSocialBoost()        => SocialBoost = true;
    public void StampPromotion(DateTimeOffset until) { IsPromoted = true; PromotedUntil = until; }

    // SEO update (separate PATCH endpoint)
    public void UpdateSeo(string? metaTitle, string? metaDescription) { ... }

    // Counter updates (called by interaction events)
    public void IncrementLikeCount()      => LikeCount++;
    public void DecrementLikeCount()      => LikeCount = Math.Max(0, LikeCount - 1);
    public void IncrementCommentCount()   => CommentCount++;
    public void DecrementCommentCount()   => CommentCount = Math.Max(0, CommentCount - 1);
    public void IncrementShareCount()     => ShareCount++;
    public void IncrementBookmarkCount()  => BookmarkCount++;
    public void DecrementBookmarkCount()  => BookmarkCount = Math.Max(0, BookmarkCount - 1);
}
```

```csharp
public class VideoEntity : Aggregate<Guid>

    public string? ThumbnailStorageKey   { get; private set; }
    public string? YoutubeVideoId        { get; private set; }   // required gate before Published
    public bool    SocialBoost           { get; private set; }
    public bool    IsPromoted            { get; private set; }
    public DateTimeOffset? PromotedUntil { get; private set; }

    public string? MetaDescription        { get; private set; }

    // Denormalized
    public decimal RatingAverage  { get; private set; }
    public int     RatingCount    { get; private set; }
    public int     ShareCount     { get; private set; }

    // Navigation
    public CustomerEntity?         Customer   { get; private set; }
    public ContentOrderItemEntity? OrderItem  { get; private set; }
    public CategoryEntity          Category   { get; private set; } = null!;
    public ICollection<VideoTagEntity>    Tags    { get; } = new List<VideoTagEntity>();
    public ICollection<VideoRatingEntity> Ratings { get; } = new List<VideoRatingEntity>();
    public ICollection<VideoShareEntity>  Shares  { get; } = new List<VideoShareEntity>();
    public ICollection<ShortVideoEntity>  Shorts  { get; } = new List<ShortVideoEntity>();

    private VideoEntity() { }

    public static VideoEntity CreateFree(Guid id, Guid categoryId,
        string title, string slug, string authorId, string? description) { ... }

    public static VideoEntity CreatePaid(Guid id, Guid customerId, Guid orderItemId,
        Guid categoryId, string title, string slug, string authorId, string? description) { ... }

    // Called when thumbnail is uploaded or replaced via POST /admin/videos/{id}/images.
    // ThumbnailStorageKey is the provider-agnostic identifier used to delete the old
    // asset from media storage when the thumbnail is replaced or the video is hard deleted.
    public void UpdateThumbnail(string thumbnailUrl, string thumbnailStorageKey) { ... }

    // Same status transitions as ArticleEntity
    public void Submit()             => Status = EnumContentStatus.PendingPayment;
    public void MarkPendingReview()  => Status = EnumContentStatus.PendingReview;
    public void Approve()            => Status = EnumContentStatus.Approved;

    // Publish requires YoutubeVideoId — enforced here
    public void Publish()
    {
        if (string.IsNullOrWhiteSpace(YoutubeVideoId))
            throw ContentErrors.VideoRequiresYoutubeLink();
        Status = EnumContentStatus.Published;
        PublishedAt = DateTimeOffset.UtcNow;
    }

    public void Reject(string reason)  { Status = EnumContentStatus.Rejected; RejectionReason = reason; }
    public void Archive()              => Status = EnumContentStatus.Archived;

    // Sets YoutubeVideoId only. The handler (AttachYoutubeIdCommandHandler) is responsible
    // for downloading the YouTube thumbnail and calling UpdateThumbnail() separately.
    // Keeping these two concerns separate makes the entity method simple and testable.
    public void AttachYoutubeId(string youtubeVideoId)       => YoutubeVideoId = youtubeVideoId;

    public void UpdateThumbnail(string thumbnailUrl, string thumbnailStorageKey) { ... }

    public void MarkHasLyrics()                              => HasLyrics = true;
    public void UpdateSeo(string? metaTitle, string? metaDescription) { ... }

    // Rating recomputed by app after each video_rating insert/update
    public void UpdateRating(decimal average, int count)
    {
        RatingAverage = average;
        RatingCount   = count;
    }

    public void IncrementShareCount() => ShareCount++;
}
```

### 4.2 EF Configurations — Key Mappings

All configurations implement `IEntityTypeConfiguration<T>` and are auto-discovered by
`ApplyConfigurationsFromAssembly`. Only the non-obvious mappings are highlighted below.

#### Enum → string conversions (all enums)

```csharp
builder.Property(x => x.Status)
    .HasConversion<string>()
    .IsRequired();
```

EF's default `HasConversion<string>()` uses the enum member name (e.g. `"PendingPayment"`).
The PostgreSQL enum values in the SQL schema use snake_case (`pending_payment`). Two options:
- Store as the C# enum name string and accept the deviation from the SQL schema, OR
- Use a custom `ValueConverter` that lowercases and snake_cases the string

> **Recommendation**: use a shared `SnakeCaseEnumConverter<T>` helper that converts
> `PendingPayment` → `pending_payment` to stay consistent with the SQL schema.

#### ArticleConfiguration — notable mappings

```csharp
// Enum stored as snake_case string
builder.Property(a => a.Status)
    .HasConversion(new SnakeCaseEnumConverter<EnumContentStatus>())
    .IsRequired();

// order_item_id: nullable FK (set AFTER content_order_items row exists)

builder.HasOne(a => a.OrderItem)
    .WithOne(oi => oi.Article)
    .HasForeignKey<ArticleEntity>(a => a.OrderItemId)
    .OnDelete(DeleteBehavior.SetNull);

// customer_id: nullable FK
builder.HasOne(a => a.Customer)
    .WithMany(c => c.Articles)
    .HasForeignKey(a => a.CustomerId)
    .OnDelete(DeleteBehavior.SetNull);

// Counters default to 0
builder.Property(a => a.LikeCount).HasDefaultValue(0).IsRequired();
// ... same for all counters

// Indexes
builder.HasIndex(a => a.Slug).IsUnique();
builder.HasIndex(a => a.Status).HasDatabaseName("ix_articles_status");
builder.HasIndex(a => a.PublishedAt)
    .HasFilter("status = 'published'")
    .HasDatabaseName("ix_articles_published_at");
builder.HasIndex(new[] { nameof(ArticleEntity.IsPromoted), nameof(ArticleEntity.PromotedUntil) })
    .HasFilter("is_promoted = true")
    .HasDatabaseName("ix_articles_promoted");
builder.HasIndex(a => a.CategoryId).HasDatabaseName("ix_articles_category_id");
builder.HasIndex(a => a.CustomerId).HasDatabaseName("ix_articles_customer_id");
builder.HasIndex(a => a.OrderItemId)
    .HasFilter("order_item_id IS NOT NULL")
    .HasDatabaseName("ix_articles_order_item_id");
```

## 6. Migration Strategy

### Initial migration (creates entire schema)

```bash
dotnet ef migrations add InitContentSchema \
  --project src/Modules/Content \
  --startup-project src/Api \
  --context ContentDbContext
```

### Run migration

```bash
dotnet ef database update \
  --project src/Modules/Content \
  --startup-project src/Api \
  --context ContentDbContext
```

### What the migration creates

The generated migration must produce:
1. `CREATE SCHEMA IF NOT EXISTS content`
2. All 28 tables in the correct dependency order (lookup → catalog → editorial → commerce → interactions)
3. All indexes (status, slug, promoted, category, customer, order_item — including filtered/partial indexes)
4. The `ALTER TABLE` back-references for `articles.order_item_id` and `videos.order_item_id`

> EF Core generates the `ALTER TABLE` statements automatically from the `HasOne` / `HasForeignKey`
> configurations — the order they appear in the migration is handled by EF's dependency resolver.
